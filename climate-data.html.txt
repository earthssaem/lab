// ===== 공통: 선형회귀 =====
function linearRegression(xs,ys){
  const n=xs.length;
  let sx=0,sy=0,sxy=0,sx2=0;
  for(let i=0;i<n;i++){sx+=xs[i];sy+=ys[i];sxy+=xs[i]*ys[i];sx2+=xs[i]*xs[i];}
  const slope=(n*sxy-sx*sy)/(n*sx2-sx*sx);
  const intercept=(sy-slope*sx)/n;
  return{slope,intercept};
}

  const idx=parseInt(document.getElementById('gasSelect').value);
  const {years,vals}=getGasSeriesData(idx);
  const fy=[],fv=[];
  years.forEach((y,i)=>{if(y>=sy&&y<=ey){fy.push(y);fv.push(vals[i]);}});
  if(fy.length<2)return;
  const r=linearRegression(fy,fv);gSlope=r.slope;gIntercept=r.intercept;
  const tArr=years.map(y=>(y>=sy&&y<=ey)?parseFloat((gSlope*y+gIntercept).toFixed(3)):null);
  drawGasChart(tArr);
  const dir=gSlope>0?'증가':'감소';
  const rb=document.getElementById('gasTrendResult');rb.style.display='block';rb.className='result-box';
  rb.innerHTML='<b>'+sy+'년 ~ '+ey+'년</b> 분석 결과<br>'+GAS_NAMES[idx]+' 농도는 매년 평균 <span class="highlight">'+Math.abs(gSlope).toFixed(4)+' '+GAS_UNITS[idx]+'/년</span> 씩 <b>'+dir+'</b> 중입니다.<br><span style="font-size:12px;color:#94a3b8">10년 누적 변화: '+Math.abs(gSlope*10).toFixed(2)+' '+GAS_UNITS[idx]+'</span>';
  document.getElementById('gasPredictResult').style.display='none';
}

function predictGas(){
  const ty=parseInt(document.getElementById('gTargetYear').value);
  const idx=parseInt(document.getElementById('gasSelect').value);
  const rb=document.getElementById('gasPredictResult');rb.style.display='block';
  if(!gSlope){rb.className='result-box warn';rb.innerHTML='⚠️ 미션 1의 추세 분석을 먼저 실행해주세요!';return;}
  if(!ty){rb.className='result-box warn';rb.innerHTML='⚠️ 예측할 연도를 입력해주세요!';return;}
  rb.className='result-box green';
  const pt=(gSlope*ty+gIntercept).toFixed(2);
  rb.innerHTML='<b>'+ty+'년</b> 예상 '+GAS_NAMES[idx]+' 농도: <span class="highlight">'+pt+' '+GAS_UNITS[idx]+'</span><br><span style="font-size:12px;color:#94a3b8">선형 외삽 예측값 (위 분석 구간 변화율 적용)</span>';
}
