'use strict';
const D2R = Math.PI / 180;
const rad  = d => d * D2R;

/* ══ 평형 조석론 ══ */
function calcAmps(lat, decl) {
  const phi = rad(lat), delta = rad(decl);
  return { As: Math.cos(phi)**2 * Math.cos(delta)**2,
           Ad: Math.sin(2*phi) * Math.sin(2*delta) };
}
function calcTide(t, As, Ad) {
  return As * Math.cos(4*Math.PI*t/24) + Ad * Math.cos(2*Math.PI*t/24);
}
function getTideClass(As, Ad) {
  const F = Math.abs(Ad) / (Math.abs(As) + 1e-9);
  
  // 해양학의 표준 폼 지수(Form Factor, F) 분류 기준 적용
  if (F < 0.25) return { txt:'반일주조 (Semi-diurnal)', cls:'type-semi' };
  if (F < 3.0)  return { txt:'혼합조 (Mixed)',          cls:'type-mixed' };
  return        { txt:'일주조 (Diurnal)',               cls:'type-di' };
}

  return             { txt:`현재 위도: ${lat}° — 고위도 A 구역`, cls:'zone-a-info' };
}

/* ══ 3D→2D 투영 ══ */
function project(latDeg, lonDeg, theta, cx, cy, R) {
  const phi   = rad(latDeg);
  const angle = rad(lonDeg) + theta;
  return {
    sx:    cx + R * Math.cos(phi) * Math.sin(angle),
    sy:    cy - R * Math.sin(phi),
    front: Math.cos(angle) > 0,
    cosA:  Math.cos(angle)
  };
}

  const eg=vc.createRadialGradient(cx-R*0.22,cy-R*0.22,R*0.05,cx,cy,R);
  eg.addColorStop(0,'#5bbef8'); eg.addColorStop(0.5,'#2a7dc0'); eg.addColorStop(1,'#1a3d72');
  vc.beginPath(); vc.arc(cx,cy,R,0,2*Math.PI); vc.fillStyle=eg; vc.fill();

  /* 6대륙 */
  vc.save();
  vc.beginPath(); vc.arc(cx,cy,R-1,0,2*Math.PI); vc.clip();
  CONTINENTS.forEach(cont => {
    const cp = project(cont.center[0], cont.center[1], theta, cx, cy, R);
    if (!cp.front) return;
    const pp = cont.pts.map(pt => project(pt[0], pt[1], theta, cx, cy, R));
    vc.beginPath();
    vc.moveTo(pp[0].sx, pp[0].sy);
    for (let i=1; i<pp.length; i++) {
      const prev=pp[i-1], cur=pp[i];
      vc.quadraticCurveTo(prev.sx, prev.sy, (prev.sx+cur.sx)/2, (prev.sy+cur.sy)/2);
    }
    vc.closePath();
    vc.fillStyle=cont.color; vc.fill();
    vc.strokeStyle='rgba(40,100,40,0.35)'; vc.lineWidth=0.7; vc.stroke();
  });
  vc.restore();

  /* 지구 테두리 */
  vc.beginPath(); vc.arc(cx,cy,R,0,2*Math.PI);
  vc.strokeStyle='rgba(255,255,255,0.20)'; vc.lineWidth=1.5; vc.stroke();

  /* 간조선 앞면 — 지구·대륙 위에 실선으로 선명하게 */
  vc.save();
  vc.beginPath(); vc.arc(cx,cy,R,0,2*Math.PI); vc.clip();
  vc.beginPath();
  vc.moveTo(cx+R*Math.cos(perpAngle), cy+R*Math.sin(perpAngle));
  vc.lineTo(cx-R*Math.cos(perpAngle), cy-R*Math.sin(perpAngle));
  vc.strokeStyle='rgba(220,38,38,0.80)'; vc.lineWidth=1.4; vc.stroke();
  vc.restore();

  /* ★ 수정: 자전축 — 검정색 계열로 변경 */
  const axTop=cy-R*1.28, axBot=cy+R*1.28;
  vc.save(); vc.setLineDash([5,4]);
  vc.beginPath(); vc.moveTo(cx,axTop); vc.lineTo(cx,axBot);
  vc.strokeStyle='rgba(80,80,80,0.60)'; vc.lineWidth=1.8; vc.stroke();
  vc.setLineDash([]); vc.restore();

  /* ★ 수정: N·S 라벨 — 검정색 + 흰 테두리로 가독성 확보 */
  const axFs = Math.round(R*0.14);
  vc.font=`bold ${axFs}px sans-serif`;
  vc.textAlign='center';
  /* 흰 테두리 */
  vc.strokeStyle='rgba(255,255,255,0.9)'; vc.lineWidth=3;
  vc.strokeText('N', cx, axTop-7);
  vc.strokeText('S', cx, axBot+R*0.19);
  /* 검정 텍스트 */
  vc.fillStyle='rgba(70,70,70,0.85)';
  vc.fillText('N', cx, axTop-7);
  vc.fillText('S', cx, axBot+R*0.19);

  /* 자전 방향 화살표 제거 */

  /* 적도선 */
  vc.beginPath(); vc.moveTo(cx-R,cy); vc.lineTo(cx+R,cy);
  vc.strokeStyle='rgba(255,255,255,0.38)'; vc.lineWidth=1.4; vc.stroke();
  vc.font=`${Math.round(R*0.10)}px 'Noto Sans KR',sans-serif`;
  vc.textAlign='right';
  vc.fillStyle='rgba(70,70,70,0.80)';
  vc.fillText('적도', cx-R-5, cy+4);

  /* 관측자 위도 타원 */
  if (latDeg > 0) {
    const phi=rad(latDeg), hw=R*Math.cos(phi), vr=hw*0.15;
    const oy=cy-R*Math.sin(phi);
    vc.save(); vc.setLineDash([4,3]);
    vc.beginPath(); vc.ellipse(cx,oy,hw,vr,0,0,2*Math.PI);
    vc.strokeStyle='rgba(239,68,68,0.50)'; vc.lineWidth=1.3; vc.stroke();
    vc.setLineDash([]); vc.restore();
    vc.fillStyle='rgba(239,68,68,0.80)';
    vc.font=`${Math.round(R*0.10)}px 'Noto Sans KR',sans-serif`;
    vc.textAlign='right';
    vc.fillText(`위도 ${latDeg}°`, cx-hw-5, oy+4);
  }

  /* 관측자 (서태평양 경도 130°) */
  {
    const OBS_LON=45;  /* t=0일 때 앞면에 위치 */
    const phi_obs=rad(latDeg);
    const angle_obs=rad(OBS_LON)+theta;
    const cosA_obs=Math.cos(angle_obs);
    const sinA_obs=Math.sin(angle_obs);
    const hw_obs=R*Math.cos(phi_obs);
    const vr_obs=hw_obs*0.15;
    const oy_obs=cy-R*Math.sin(phi_obs);
    const obsX=cx+hw_obs*sinA_obs;
    const obsY=oy_obs+vr_obs*cosA_obs;
    const dx=obsX-cx, dy=obsY-cy, dl=Math.sqrt(dx*dx+dy*dy)||1;
    const nx=dx/dl, ny=dy/dl, tx=-ny, ty=nx;
    const isFront=cosA_obs>0;
    const alpha=isFront?1.0:0.3;
    const scale=isFront?1.0:0.7;
    const s=R*0.075*scale;
    const col=`rgba(239,68,68,${alpha.toFixed(2)})`;
    const wht=`rgba(255,255,255,${(alpha*0.9).toFixed(2)})`;

    const {As,Ad}=calcAmps(latDeg,declDeg);
    const Ht=calcTide(simTime,As,Ad);
    const barPx=Ht*R*0.22;
    const topX=obsX+nx*s*3.1, topY=obsY+ny*s*3.1;
    const barA=Math.min(1.0,alpha+0.2);
    vc.beginPath(); vc.moveTo(topX,topY); vc.lineTo(topX,topY-barPx);
    if(!isFront) vc.setLineDash([3,2]);
    vc.strokeStyle=Ht>=0?`rgba(37,99,235,${barA.toFixed(2)})`:`rgba(245,158,11,${barA.toFixed(2)})`;
    vc.lineWidth=2.2; vc.stroke(); vc.setLineDash([]);
    if(isFront){
      vc.fillStyle=Ht>=0?'rgba(29,78,216,0.90)':'rgba(180,83,9,0.90)';
      vc.font=`bold ${Math.round(R*0.095)}px 'DM Mono',monospace`;
      vc.textAlign='center';
      vc.fillText((Ht>=0?'+':'')+Ht.toFixed(2)+'m',topX,topY-barPx-R*0.06);
    }
    vc.beginPath(); vc.arc(obsX+1,obsY+1,s*0.85,0,2*Math.PI);
    vc.fillStyle=`rgba(0,0,0,${(alpha*0.15).toFixed(2)})`; vc.fill();
    vc.beginPath(); vc.moveTo(obsX,obsY); vc.lineTo(obsX+nx*s*1.4,obsY+ny*s*1.4);
    if(!isFront) vc.setLineDash([3,2]);
    vc.strokeStyle=col; vc.lineWidth=s*0.55; vc.stroke(); vc.setLineDash([]);
    const ax=obsX+nx*s*0.9, ay=obsY+ny*s*0.9;
    vc.beginPath(); vc.moveTo(ax-tx*s*0.8,ay-ty*s*0.8); vc.lineTo(ax+tx*s*0.8,ay+ty*s*0.8);
    if(!isFront) vc.setLineDash([3,2]);
    vc.strokeStyle=col; vc.lineWidth=s*0.45; vc.stroke(); vc.setLineDash([]);
    const hx=obsX+nx*s*2.2, hy=obsY+ny*s*2.2;
    vc.beginPath(); vc.arc(hx,hy,s*0.85,0,2*Math.PI);
    vc.fillStyle=col; vc.fill();
    vc.strokeStyle=wht; vc.lineWidth=s*0.28; vc.stroke();
    if(isFront){
      vc.fillStyle='rgba(220,38,38,0.90)';
      vc.font=`bold ${Math.round(R*0.10)}px 'Noto Sans KR',sans-serif`;
      vc.textAlign='left';
      vc.fillText('관측자',obsX+R*0.10,obsY-R*0.04);
    }
  }

  /* 달 */
  {
    vc.save(); vc.setLineDash([5,4]);
    vc.beginPath();
    vc.moveTo(moonX-moonR*0.9,moonY);
    vc.lineTo(cx+seaA*Math.cos(moonAngle-Math.PI),cy+seaA*Math.sin(moonAngle-Math.PI));
    vc.strokeStyle='rgba(245,158,11,0.28)'; vc.lineWidth=1.5; vc.stroke();
    vc.setLineDash([]); vc.restore();
    const mg=vc.createRadialGradient(moonX-moonR*0.3,moonY-moonR*0.3,moonR*0.08,moonX,moonY,moonR);
    mg.addColorStop(0,'#fde68a'); mg.addColorStop(1,'#d97706');
    vc.beginPath(); vc.arc(moonX,moonY,moonR,0,2*Math.PI);
    vc.fillStyle=mg; vc.fill(); vc.strokeStyle='#b45309'; vc.lineWidth=1; vc.stroke();
    vc.fillStyle='#92400e';
    vc.font=`bold ${Math.round(R*0.12)}px 'Noto Sans KR',sans-serif`;
    vc.textAlign='center';
    vc.fillText('달',moonX,moonY+moonR+R*0.13);
    vc.font=`${Math.round(R*0.095)}px 'DM Mono',monospace`;
    vc.fillText(`적위 ${declDeg.toFixed(1)}°`,moonX,moonY+moonR+R*0.24);
    const dirAng=Math.atan2(moonY-cy,moonX-cx);
    const arrLen=R*0.32;
    const arrSX=cx+(seaA+R*0.05)*Math.cos(dirAng);
    const arrSY=cy+(seaA+R*0.05)*Math.sin(dirAng);
    const arrEX=arrSX+arrLen*Math.cos(dirAng);
    const arrEY=arrSY+arrLen*Math.sin(dirAng);
    vc.beginPath(); vc.moveTo(arrSX,arrSY); vc.lineTo(arrEX,arrEY);
    vc.strokeStyle='#d97706'; vc.lineWidth=2.0; vc.stroke();
    vc.save(); vc.translate(arrEX,arrEY); vc.rotate(dirAng);
    vc.beginPath(); vc.moveTo(0,0); vc.lineTo(-10,-4); vc.lineTo(-10,4);
    vc.closePath(); vc.fillStyle='#d97706'; vc.fill(); vc.restore();
    vc.fillStyle='#92400e';
    vc.font=`bold ${Math.round(R*0.105)}px 'Noto Sans KR',sans-serif`;
    vc.textAlign='center';
    vc.fillText('달의 방향',arrEX-Math.sin(dirAng)*R*0.14,arrEY+Math.cos(dirAng)*R*0.14);
  }
}

/* ══ 애니메이션 ══ */
function tick(ts){
  if(!lastTs) lastTs=ts;
  const dt=Math.min((ts-lastTs)/1000,0.1); lastTs=ts;
  simTime=(simTime+dt*1.2)%24;
  syncTimeSlider(); updateAll();
  if(playing) rafId=requestAnimationFrame(tick);
}
function togglePlay(){
  playing=!playing;
  const btn=document.getElementById('playBtn');
  if(playing){ btn.textContent='⏸ 일시정지'; lastTs=null; rafId=requestAnimationFrame(tick); }
  else{ btn.textContent='▶ 재생'; cancelAnimationFrame(rafId); }
}
function resetSim(){
  playing=false; cancelAnimationFrame(rafId);
  simTime=0; lastTs=null;
  syncTimeSlider(); document.getElementById('playBtn').textContent='▶ 재생'; updateAll();
}
function syncTimeSlider(){
  const sl=document.getElementById('timeSlider');
  sl.value=simTime; setSliderFill(sl);
  document.getElementById('timeVal').textContent=simTime.toFixed(1)+' h';
  const hh=Math.floor(simTime).toString().padStart(2,'0');
  const mm=Math.floor((simTime%1)*60).toString().padStart(2,'0');
  document.getElementById('timeDisplay').textContent=hh+':'+mm;
}
function setSliderFill(el){ el.style.setProperty('--pct',((el.value-el.min)/(el.max-el.min)*100)+'%'); }
function updateAll(){ drawViz(); updateChart(); }
